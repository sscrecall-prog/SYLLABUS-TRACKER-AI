121:     // Spaced Repetition Logic
122:     suspend fun markChapterRevised(chapter: SyllabusItem) = withContext(Dispatchers.IO) {
123:         val settings = settingsDao.getSettingsDirect() ?: AppSettings()
124:         val intervals = settings.revisionIntervals
125:         val currentRevCount = chapter.revisionCount + 1
126:         
127:         val intervalIndex = (currentRevCount - 1).coerceAtMost(intervals.size - 1)
128:         val daysToAdd = intervals[intervalIndex]
129:         val now = System.currentTimeMillis()
130:         val nextDate = now + (daysToAdd.toLong() * 24 * 60 * 60 * 1000L)
131: 
132:         val updated = chapter.copy(
133:             revisionCount = currentRevCount,
134:             lastStudiedTimestamp = now,
135:             nextRevisionTimestamp = nextDate,
136:             status = if (currentRevCount >= 5) ChapterStatus.MASTERED else ChapterStatus.COMPLETED,
137:             completionPercentage = 100
138:         )
139:         syllabusDao.updateItem(updated)
140:     }

151:     suspend fun updateChapterStatus(chapter: SyllabusItem, newStatus: ChapterStatus) = withContext(Dispatchers.IO) {
152:         val now = System.currentTimeMillis()
153:         val settings = settingsDao.getSettingsDirect() ?: AppSettings()
154:         val intervals = settings.revisionIntervals
155: 
156:         val updated = when (newStatus) {
157:             ChapterStatus.COMPLETED -> {
158:                 val nextRev = if (chapter.nextRevisionTimestamp == null || chapter.nextRevisionTimestamp <= now) {
159:                     now + (intervals.firstOrNull() ?: 1) * 24L * 60 * 60 * 1000L
160:                 } else chapter.nextRevisionTimestamp
161:                 chapter.copy(
162:                     status = newStatus,
163:                     completionPercentage = 100,
164:                     lastStudiedTimestamp = now,
165:                     nextRevisionTimestamp = nextRev
166:                 )
167:             }
168:             ChapterStatus.MASTERED -> {
169:                 chapter.copy(status = newStatus, completionPercentage = 100, confidence = 5)
170:             }
171:             ChapterStatus.NOT_STARTED -> {
172:                 chapter.copy(status = newStatus, completionPercentage = 0)
173:             }
174:             ChapterStatus.WEAK -> {
175:                 chapter.copy(status = newStatus, confidence = chapter.confidence.coerceAtMost(2))
176:             }
177:             ChapterStatus.REVISION_DUE -> {
178:                 chapter.copy(status = newStatus, nextRevisionTimestamp = now)
179:             }
180:             else -> {
181:                 chapter.copy(status = newStatus)
182:             }
183:         }
184:         syllabusDao.updateItem(updated)
185:     }
